Reading a file in two encodings

def Read_Text_File(filename):
    try:
        f = open(filename, "r", encoding="utf-8")
    except:
        raise FileNotFoundError
    try:
        text = f.read()
    except UnicodeDecodeError:
        f.close()
        f = open(filename, "r",
             encoding="windows-1252")
        text = f.read()
    f.close()
    return text
